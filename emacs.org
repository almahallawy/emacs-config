(defun c-lineup-arglist-tabs-only (ignored)
  "Line up argument lists by tabs, not spaces"
  (let* ((anchor (c-langelem-pos c-syntactic-element))
	 (column (c-langelem-2nd-pos c-syntactic-element))
	 (offset (- (1+ column) anchor))
	 (steps (floor offset c-basic-offset)))
    (* (max steps 1)
       c-basic-offset)))

** Copilot session token reset
The Copilot session token is bound to the public IP.  A network change (e.g.
VPN on/off) changes it, so a cached token can start returning HTTP 403
before it expires.  gptel only renews on expiry, so clear it manually.
#+BEGIN_SRC emacs-lisp
(defun kia/gptel-gh-reset-token ()
  "Discard the cached Copilot session token so gptel fetches a new one.
The token is bound to the public IP; run this after switching VPN on/off
if requests fail with HTTP 403."
  (interactive)
  (dolist (b (mapcar #'cdr gptel--known-backends))
    (when (gptel--gh-p b)
      (setf (gptel--gh-token b) nil)))
  (when (file-exists-p gptel-gh-token-file)
    (delete-file gptel-gh-token-file))
  (message "Copilot session token cleared; next request will fetch a new one."))
#+END_SRC

** Agent unleash (per-buffer tool confirmation toggle)
Toggle tool-call confirmation in the current gptel buffer only, via a
buffer-local `gptel-pre-tool-call-functions' hook.  Other buffers unaffected.
C-c C-u in a gptel buffer toggles it.
#+BEGIN_SRC emacs-lisp
(defun my-gptel-agent-unleash--hook (_call)
  "Skip confirmation for every tool call."
  '(:confirm nil))

(defun my-gptel-agent-unleash ()
  "Toggle tool-call confirmation in the current gptel buffer only."
  (interactive)
  (if (memq #'my-gptel-agent-unleash--hook gptel-pre-tool-call-functions)
      (progn (remove-hook 'gptel-pre-tool-call-functions
                          #'my-gptel-agent-unleash--hook t)
             (message "gptel agent reined in (%s)" (buffer-name)))
    (add-hook 'gptel-pre-tool-call-functions
              #'my-gptel-agent-unleash--hook nil t)
    (message "gptel agent unleashed (%s)" (buffer-name))))

(keymap-set gptel-mode-map "C-c C-u" #'my-gptel-agent-unleash)
#+END_SRC
