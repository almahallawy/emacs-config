* Generate the emacs configuration file
 M-x org-babel-tangle (C-c C-v t)

This will generate: kia-config.el and my-gptel.el (my-gptel.el is
byte-compiled automatically once this config is loaded).
Add the following to your ~/.emacs file (keep Custom's block below it):

(load "~/src/emacs-config/.emacs")

The repo's .emacs (src/emacs-config is the location of the emacs config) contains:

;; Load a newer .el over a stale .elc (e.g. my-gptel after tangling).
(setq load-prefer-newer t)
(defvar kia-emacs-directory "~/src/emacs-config")
(add-to-list 'load-path kia-emacs-directory)
(require 'kia-config)

* Prerequisites
Clone these into ~/src before the first start; they are loaded from there:
- https://github.com/karthink/gptel.git          -> ~/src/gptel
- https://github.com/karthink/gptel-agent.git    -> ~/src/gptel-agent
- https://github.com/jeremy-compostella/cscope   -> ~/src/cscope

System packages:
- pandoc (markdown-mode preview/export): sudo apt install pandoc
- pdf-tools build deps (epdfinfo is compiled on first load, see PDF):
  sudo apt install libpoppler-glib-dev libpoppler-private-dev libpng-dev
  zlib1g-dev autoconf automake gcc make pkg-config

Shell (not tangled, manual step):
- Append the ~/.bashrc snippet from Shell > Directory tracking (OSC 7) to
  ~/.bashrc.  Without it M-x shell has no prompt colours, no directory
  tracking (default-directory and F1 buffer names don't follow cd),
  no colour from git/ls/grep, and pagers (less) break.

(defun c-lineup-arglist-tabs-only (ignored)
  "Line up argument lists by tabs, not spaces"
  (let* ((anchor (c-langelem-pos c-syntactic-element))
	 (column (c-langelem-2nd-pos c-syntactic-element))
	 (offset (- (1+ column) anchor))
	 (steps (floor offset c-basic-offset)))
    (* (max steps 1)
       c-basic-offset)))

** my-gptel.el (personal gptel extensions)
The following sections tangle to a separate file, my-gptel.el, which is
loaded here.  It is generated, like kia-config.el, and byte-compiled
automatically after every tangle (C-c C-v t), see the hook below.
.emacs sets load-prefer-newer, so a stale my-gptel.elc is never loaded
over a newer my-gptel.el.
Manual compile (e.g. tangling from a bare emacs --batch):
  emacs --batch -L ~/src/gptel -L ~/src/gptel-agent -f batch-byte-compile ~/src/emacs-config/my-gptel.el
#+BEGIN_SRC emacs-lisp
(require 'my-gptel)
;; Optional private extensions: load any my-gptel-*.el tangled from other org
;; files in this directory (none present = nothing loaded).
(dolist (f (directory-files kia-emacs-directory nil "\\`my-gptel-.+\\.el\\'"))
  (require (intern (file-name-sans-extension f)) nil t))

(defun my-gptel-compile-after-tangle ()
  "Byte-compile my-gptel*.el right after it is tangled.
Runs in the tangled file's buffer (`org-babel-post-tangle-hook')."
  (when (and buffer-file-name
             (string-match-p "\\`my-gptel.*\\.el\\'"
                             (file-name-nondirectory buffer-file-name)))
    (byte-compile-file buffer-file-name)))

** Copilot session token reset
The Copilot session token is bound to the public IP.  A network change (e.g.
VPN on/off) changes it, so a cached token can start returning HTTP 403
before it expires.  gptel only renews on expiry, so clear it manually.
#+BEGIN_SRC emacs-lisp :tangle my-gptel.el
(defun my-gptel-gh-reset-token ()
  "Discard the cached Copilot session token so gptel fetches a new one.
The token is bound to the public IP; run this after switching VPN on/off
if requests fail with HTTP 403."
  (interactive)
  (dolist (b (mapcar #'cdr gptel--known-backends))
    (when (gptel--gh-p b)
      (setf (gptel--gh-token b) nil)))
  (when (file-exists-p gptel-gh-token-file)
    (delete-file gptel-gh-token-file))
  (message "Copilot session token cleared; next request will fetch a new one."))
#+END_SRC

** Agent unleash (per-buffer tool confirmation toggle)

  '(:confirm nil))

(defun my-gptel-agent-unleash ()
  "Toggle tool-call confirmation in the current gptel buffer only."
  (interactive)
  (if (memq #'my-gptel-agent-unleash--hook gptel-pre-tool-call-functions)

                          #'my-gptel-agent-unleash--hook t)

  "Describe request STATE with request INFO in words."
  (pcase state
    ('WAIT "waiting for LLM")
    ('TYPE "receiving")
    ('TOOL (if (plist-get info :tool-pending) "waiting C-c C-c" "running tools"))
    ('DONE "done")
    ('ERRS "error")
    ('ABRT "aborted")
    ('IDLE "idle")

  (with-current-buffer buf
    (let* ((fsm (or (cl-find buf active :key #'my-gptel-sessions--fsm-buffer)
                    gptel--fsm-last))
           (state (if fsm (gptel-fsm-state fsm) 'IDLE))
           (info (and fsm (gptel-fsm-info fsm)))
           (running (and fsm (memq fsm active))))
      (list buf
            (vector (buffer-name)

(defun my-gptel-sessions--entries ()
  "Return entries for all gptel buffers and buffers with running requests.
Running sub-agent requests follow their parent buffer's row."
  (let* ((all (mapcar #'cadr gptel--request-alist))
         (subs (seq-filter #'my-gptel-sessions--subagent-ov all))
         (active (seq-difference all subs))
         (bufs (seq-uniq
                (append (seq-filter (lambda (b) (buffer-local-value 'gptel-mode b))
                                    (buffer-list))
                        (delq nil (mapcar #'my-gptel-sessions--fsm-buffer all))))))
    (mapcan (lambda (b)
              (cons (my-gptel-sessions--entry b active)
                    (mapcar #'my-gptel-sessions--subagent-entry
                            (seq-filter (lambda (f)
                                          (eq (my-gptel-sessions--fsm-buffer f) b))
                                        subs))))
            (sort (seq-filter #'buffer-live-p bufs)
                  (lambda (a b) (string< (buffer-name a) (buffer-name b)))))))

~/.bashrc (OSC 7 adapted from Mickey's `myprompt'; the INSIDE_EMACS guard is ours.
Pager part from Prot's .bashrc, quoted in prot-emacs-essentials.el "Standard
Unix Shell" section: only when TERM=dumb, so vterm/term keep using less.

#+begin_src sh :tangle no
if [ -n "$INSIDE_EMACS" ] || [ "$TERM" = "dumb-emacs-ansi" ]; then
    # Ours: under ssh, report user@IP we connected to (works without DNS
    # or ~/.ssh/config); hostname after # is only for the buffer name.
    __emacs_osc7() {
        local host=$HOSTNAME
        [ -n "$SSH_CONNECTION" ] && { local c=($SSH_CONNECTION); host="$USER@${c[2]}"; }
        printf "\e]7;file://%s%s#%s\e\\" "$host" "$PWD" "$HOSTNAME"
    }
    PROMPT_COMMAND="__emacs_osc7${PROMPT_COMMAND:+;$PROMPT_COMMAND}"
    if [ "$TERM" = "dumb" ] || [ "$TERM" = "dumb-emacs-ansi" ]; then
        export PAGER=cat
        export GIT_PAGER=cat
        export TERM=dumb-emacs-ansi
        export COLORTERM=1
        # Ours: uutils ls (Ubuntu default) refuses color when TERM=dumb*,
        # even with --color=always; GNU ls (gnuls) doesn't.
        command -v gnuls >/dev/null && alias ls='gnuls --color=auto'
        # Ours: tmux refuses dumb-emacs-ansi (terminfo has hc, "hardcopy");
        # plain dumb works for ls/new -d/kill-session.  Attach still needs
        # a real terminal (vterm/Ptyxis).
        tmux() { TERM=dumb command tmux "$@"; }
        # Ours: Debian PS1 colours only for xterm/256color; force it here.
        PS1='${debian_chroot:+($debian_chroot)}\[\033[01;32m\]\u@\h\[\033[00m\]:\[\033[01;34m\]\w\[\033[00m\]\$ '
    fi
fi

** Remote shells: OSC 7 -> TRAMP path
Mickey's workflow: ssh inside M-x shell + Tramp for files.  But the built-in
`ansi-osc-directory-tracker' ignores OSC 7 from other hosts, so after
`ssh host' the buffer's `default-directory' stays local.
- Ours: map file://HOST/PATH to a TRAMP path:
  - HOST is this machine -> local PATH (as before)
  - shell started via TRAMP -> keep the prefix it started on
    (method/user/hops), swap in PATH
  - local shell where you typed `ssh user@IP' (even ssh A then ssh B)
    -> /ssh:user@IP:PATH.  The remote bashrc sends user@IP from
    $SSH_CONNECTION, so no DNS name or ~/.ssh/config entry is needed and
    known_hosts already has the IP.
  - Buffer name shows the hostname instead: *shell user@host:PATH*
    (hostname sent after # in the OSC 7 URL).
- Sets `default-directory' directly; no TRAMP connection is opened from the
  process filter (would block, maybe prompt for a password).  The connection
  is made on first use (C-x C-f, dired, TAB).
- Needs the ~/.bashrc block on the remote too.  Plain ssh does not pass
  INSIDE_EMACS, only TERM, so the block also checks TERM=dumb-emacs-ansi.

#+BEGIN_SRC emacs-lisp
(defvar-local my/shell--origin nil
  "TRAMP prefix the shell process was started on, or nil if local.")

(defun my/shell--record-origin ()
  (setq my/shell--origin (file-remote-p default-directory)))
(add-hook 'shell-mode-hook #'my/shell--record-origin)

(defvar-local my/shell--label nil
  "\"user@hostname\" of the remote the shell is on, for the buffer name.")

(defun my/ansi-osc-directory-tracker (_ text)
  "Like `ansi-osc-directory-tracker', but follow remote hosts via TRAMP."
  (let* ((url (url-generic-parse-url text))
         (user (url-user url))
         (host (url-host url))
         (path (url-unhex-string (url-filename url))))
    (when (and (equal (url-type url) "file") (file-name-absolute-p path))
      (let* ((localp (or (null host) (equal host "")
                         (string= host (downcase (system-name)))))
             (dir (cond (localp path)
                        (my/shell--origin (concat my/shell--origin path))
                        (t (format "/ssh:%s%s:%s"
                                   (if user (concat user "@") "") host path)))))
        (setq my/shell--label
              (and (not localp)
                   (concat (if user (concat user "@") "")
                           (or (url-target url) host))))
        (setq default-directory (file-name-as-directory dir))
        ;; Ours: shell.el sets this to t for shells started locally, which
        ;; disables TRAMP during completion; follow the current dir instead.
        (setq-local pcomplete-remote-file-ignore (not (file-remote-p dir)))))))

(with-eval-after-load 'ansi-osc
  (setf (alist-get "7" ansi-osc-handlers nil nil #'equal)
        #'my/ansi-osc-directory-tracker))

        ;; Ours: comint checks `last-command' to continue a search.
        (setq this-command 'comint-previous-matching-input-from-input)
        (comint-previous-matching-input-from-input arg))
    (forward-line (- arg))))

(defun my/shell-down (arg)
  "History at prompt, else move down.  Based on Prot's `prot-shell-down-dwim'."
  (interactive "^p")
  (if (comint-after-pmark-p)
      (progn
        (setq this-command 'comint-next-matching-input-from-input)
        (comint-next-matching-input-from-input arg))
    (forward-line arg)))

(with-eval-after-load 'shell
  (define-key shell-mode-map (kbd "<up>")   #'my/shell-up)
  (define-key shell-mode-map (kbd "<down>") #'my/shell-down)
  ;; Prot: easier key for clearing the buffer (default C-c M-o)
  (define-key shell-mode-map (kbd "C-c C-k") #'comint-clear-buffer))
#+END_SRC

** F1: one shell per directory
- Prot: prot-shell.el (`prot-shell', bound to F1): new shell named after
  `default-directory', reused if one exists; renamed after `cd'.
- Ours: rename on any directory change reported by OSC 7 (pushd, scripts,
  aliases), not only typed `cd ...'; applies to every shell-mode buffer.
- C-c s is taken by cscope, so F1 only (help stays on C-h).

#+BEGIN_SRC emacs-lisp
(defun my/shell--name (dir)
  (let ((d (abbreviate-file-name (directory-file-name dir))))
    ;; Ours: remote via ssh-in-shell: show user@hostname, not the TRAMP path
    (if (and (bound-and-true-p my/shell--label) (file-remote-p dir))
        (format "*shell %s:%s*" my/shell--label (file-remote-p d 'localname))
      (format "*shell %s*" d))))

(defun my/shell--rename-on-cd (_output)
  "Rename the shell buffer when `default-directory' changes."
  (let ((name (my/shell--name default-directory)))
    (unless (string-prefix-p name (buffer-name))
      (rename-buffer name t))))

(defun my/shell--track-name ()
  ;; APPEND so it runs after `comint-osc-process-output' updates the dir.
  (add-hook 'comint-output-filter-functions #'my/shell--rename-on-cd t t))

;; Ours: every shell (M-x shell too) is named after its cwd, like a
;; terminal title.  Note: M-x shell then starts a new shell each time.
(add-hook 'shell-mode-hook #'my/shell--track-name)

(defun my/shell ()
  "Shell for `default-directory', named after it; reuse if it exists.
Based on Prot's `prot-shell'."
  (interactive)
  (shell (my/shell--name default-directory)))

(global-set-key (kbd "<f1>") #'my/shell)
#+END_SRC
