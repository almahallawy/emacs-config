(defun c-lineup-arglist-tabs-only (ignored)
  "Line up argument lists by tabs, not spaces"
  (let* ((anchor (c-langelem-pos c-syntactic-element))
	 (column (c-langelem-2nd-pos c-syntactic-element))
	 (offset (- (1+ column) anchor))
	 (steps (floor offset c-basic-offset)))

       c-basic-offset)))

** Copilot session token reset
The Copilot session token is bound to the public IP.  A network change (e.g.
VPN on/off) changes it, so a cached token can start returning HTTP 403
before it expires.  gptel only renews on expiry, so clear it manually.
#+BEGIN_SRC emacs-lisp
(defun kia/gptel-gh-reset-token ()
  "Discard the cached Copilot session token so gptel fetches a new one.
The token is bound to the public IP; run this after switching VPN on/off
if requests fail with HTTP 403."
  (interactive)
  (dolist (b (mapcar #'cdr gptel--known-backends))
    (when (gptel--gh-p b)
      (setf (gptel--gh-token b) nil)))
  (when (file-exists-p gptel-gh-token-file)
    (delete-file gptel-gh-token-file))
  (message "Copilot session token cleared; next request will fetch a new one."))
#+END_SRC

** Agent unleash (per-buffer tool confirmation toggle)
Toggle tool-call confirmation in the current gptel buffer only, via a
buffer-local `gptel-pre-tool-call-functions' hook.  Other buffers unaffected.
C-c C-u in a gptel buffer toggles it.
#+BEGIN_SRC emacs-lisp :tangle my-gptel.el
(defun my-gptel-agent-unleash--hook (_call)
  "Skip confirmation for every tool call."
  '(:confirm nil))

(defun my-gptel-agent-unleash ()
  "Toggle tool-call confirmation in the current gptel buffer only."
  (interactive)
  (if (memq #'my-gptel-agent-unleash--hook gptel-pre-tool-call-functions)
      (progn (remove-hook 'gptel-pre-tool-call-functions
                          #'my-gptel-agent-unleash--hook t)
             (message "gptel agent reined in (%s)" (buffer-name)))
    (add-hook 'gptel-pre-tool-call-functions
              #'my-gptel-agent-unleash--hook nil t)
    (message "gptel agent unleashed (%s)" (buffer-name))))

(keymap-set gptel-mode-map "C-c C-u" #'my-gptel-agent-unleash)
#+END_SRC

** gptel sessions list
M-x my-gptel-sessions lists all gptel buffers and the state of their
requests, refreshed every `my-gptel-sessions-refresh-interval' seconds
while visible.  Running gptel-agent sub-agents are shown as extra rows
under their parent buffer.  Keys: RET visit, k abort, u toggle unleash,
g refresh.
#+BEGIN_SRC emacs-lisp :tangle my-gptel.el
;; gptel sessions list
(defgroup my-gptel-sessions nil
  "List gptel sessions and their request state."
  :prefix "my-gptel-sessions-"
  :group 'gptel)

(defcustom my-gptel-sessions-buffer "*gptel-sessions*"
  "Name of the gptel sessions list buffer."
  :type 'string)

(defcustom my-gptel-sessions-refresh-interval 5
  "Seconds between automatic refreshes of the sessions list.
A zero or negative value disables automatic refresh."
  :type 'number)

(defvar my-gptel-sessions--timer nil
  "Timer refreshing the sessions list.")

(defun my-gptel-sessions--fsm-buffer (fsm)
  "Return the buffer of request FSM."
  (plist-get (gptel-fsm-info fsm) :buffer))

(defun my-gptel-sessions--subagent-ov (fsm)
  "Return the gptel-agent task overlay of FSM, if it is a sub-agent."
  (let ((ov (plist-get (gptel-fsm-info fsm) :context)))
    (and (overlayp ov) (overlay-get ov 'gptel-agent) ov)))

(defun my-gptel-sessions--tag-task-overlay (orig where &optional type desc)
  "Call ORIG with WHERE, TYPE, DESC; record TYPE and DESC on the overlay."
  (let ((ov (funcall orig where type desc)))
    (overlay-put ov 'my-gptel-type type)
    (overlay-put ov 'my-gptel-desc desc)
    ov))

(with-eval-after-load 'gptel-agent-tools
  (advice-add 'gptel-agent--task-overlay :around
              #'my-gptel-sessions--tag-task-overlay))

(defun my-gptel-sessions--id-buffer (id)
  "Return the buffer of row ID (a buffer or a request FSM)."
  (if (bufferp id) id (my-gptel-sessions--fsm-buffer id)))

(defun my-gptel-sessions--abort-fsm (fsm)
  "Abort the running request FSM, like `gptel-abort' but for one request."
  (when-let* ((entry (cl-find fsm gptel--request-alist :key #'cadr))
              (info (gptel-fsm-info fsm)))
    (with-demoted-errors "Callback error: %S"
      (and-let* ((cb (plist-get info :callback))
                 ((functionp cb)))
        (funcall cb 'abort info)))
    (funcall (cddr entry))
    (setf (alist-get (car entry) gptel--request-alist nil 'remove) nil)
    (gptel--fsm-transition fsm 'ABRT)))

(defun my-gptel-sessions--status (state info)
  "Describe request STATE with request INFO in words."
  (pcase state
    ('WAIT "waiting for LLM")
    ('TYPE "receiving")
    ('TOOL (if (plist-get info :tool-pending) "waiting C-c C-c" "running tools"))
    ('DONE "done")
    ('ERRS "error")
    ('ABRT "aborted")
    ('IDLE "idle")
    (_ (symbol-name state))))

(defun my-gptel-sessions--entry (buf active)
  "Return the list entry for BUF.  ACTIVE is the list of running FSMs."
  (with-current-buffer buf
    (let* ((fsm (or (cl-find buf active :key #'my-gptel-sessions--fsm-buffer)
                    gptel--fsm-last))
           (state (if fsm (gptel-fsm-state fsm) 'IDLE))
           (info (and fsm (gptel-fsm-info fsm)))
           (running (and fsm (memq fsm active))))
      (list buf
            (vector (buffer-name)
                    (propertize (symbol-name state)
                                'face (if running 'success 'shadow))
                    (my-gptel-sessions--status state info)
                    (format "%s" (or (plist-get info :model) gptel-model))
                    (if (memq #'my-gptel-agent-unleash--hook
                              gptel-pre-tool-call-functions)
                        "yes" "")
                    (format "%s" (or (plist-get info :error) "")))))))

(defun my-gptel-sessions--subagent-entry (fsm)
  "Return the list entry for running sub-agent request FSM."
  (let* ((ov (my-gptel-sessions--subagent-ov fsm))
         (info (gptel-fsm-info fsm))
         (state (gptel-fsm-state fsm))
         (count (or (overlay-get ov 'count) 0)))
    (list fsm
          (vector (format "  ↳ %s: %s"
                          (or (overlay-get ov 'my-gptel-type) "agent")
                          (or (overlay-get ov 'my-gptel-desc) ""))
                  (propertize (symbol-name state) 'face 'success)
                  (format "%s (%d tool%s)"
                          (my-gptel-sessions--status state info)
                          count (if (= count 1) "" "s"))
                  (format "%s" (or (plist-get info :model) ""))
                  ""
                  (format "%s" (or (plist-get info :error) ""))))))

(defun my-gptel-sessions--entries ()
  "Return entries for all gptel buffers and buffers with running requests.
Running sub-agent requests follow their parent buffer's row."
  (let* ((all (mapcar #'cadr gptel--request-alist))
         (subs (seq-filter #'my-gptel-sessions--subagent-ov all))
         (active (seq-difference all subs))
         (bufs (seq-uniq
                (append (seq-filter (lambda (b) (buffer-local-value 'gptel-mode b))
                                    (buffer-list))
                        (delq nil (mapcar #'my-gptel-sessions--fsm-buffer all))))))
    (mapcan (lambda (b)
              (cons (my-gptel-sessions--entry b active)
                    (mapcar #'my-gptel-sessions--subagent-entry
                            (seq-filter (lambda (f)
                                          (eq (my-gptel-sessions--fsm-buffer f) b))
                                        subs))))
            (sort (seq-filter #'buffer-live-p bufs)
                  (lambda (a b) (string< (buffer-name a) (buffer-name b)))))))

(defun my-gptel-sessions--stop-timer ()
  "Cancel the refresh timer."
  (when (timerp my-gptel-sessions--timer)
    (cancel-timer my-gptel-sessions--timer))
  (setq my-gptel-sessions--timer nil))

(defun my-gptel-sessions--refresh ()
  "Refresh the sessions list if visible; stop the timer if it is gone."
  (if-let* ((buf (get-buffer my-gptel-sessions-buffer)))
      (when (get-buffer-window buf t)
        (with-current-buffer buf (tabulated-list-print t)))
    (my-gptel-sessions--stop-timer)))

(defun my-gptel-sessions-visit ()
  "Jump to the gptel session (or sub-agent task) at point."
  (interactive)
  (when-let* ((id (tabulated-list-get-id))
              (buf (my-gptel-sessions--id-buffer id)))
    (pop-to-buffer buf)
    (when-let* (((not (bufferp id)))
                (ov (my-gptel-sessions--subagent-ov id))
                (pos (overlay-start ov)))
      (goto-char pos))))

(defun my-gptel-sessions-abort ()
  "Abort the request (session or sub-agent) at point."
  (interactive)
  (when-let* ((id (tabulated-list-get-id))
              (buf (my-gptel-sessions--id-buffer id))
              (fsm (if (bufferp id)
                       (cl-find-if
                        (lambda (f)
                          (and (eq (my-gptel-sessions--fsm-buffer f) buf)
                               (not (my-gptel-sessions--subagent-ov f))))
                        (mapcar #'cadr gptel--request-alist))
                     id)))
    (when (y-or-n-p (format "Abort %s in %s? "
                            (if (bufferp id) "request" "sub-agent")
                            (buffer-name buf)))
      (my-gptel-sessions--abort-fsm fsm)
      (tabulated-list-print t))))

(defun my-gptel-sessions-toggle-unleash ()
  "Toggle tool-call confirmation in the gptel session at point."
  (interactive)
  (when-let* ((buf (my-gptel-sessions--id-buffer (tabulated-list-get-id))))
    (with-current-buffer buf (my-gptel-agent-unleash))
    (tabulated-list-print t)))

(defvar-keymap my-gptel-sessions-mode-map
  "RET" #'my-gptel-sessions-visit
  "k"   #'my-gptel-sessions-abort
  "u"   #'my-gptel-sessions-toggle-unleash)

(define-derived-mode my-gptel-sessions-mode tabulated-list-mode "gptel-sessions"
  "Major mode listing gptel sessions."
  ;; Not sortable: rows are ordered so sub-agents follow their parent.
  (setq tabulated-list-format [("Buffer" 40 nil) ("State" 6 nil) ("Status" 26 nil)
                               ("Model" 18 nil) ("Unleashed" 9 nil) ("Error" 30 nil)]
        tabulated-list-entries #'my-gptel-sessions--entries
        tabulated-list-sort-key nil)
  (tabulated-list-init-header)
  (add-hook 'kill-buffer-hook #'my-gptel-sessions--stop-timer nil t)
  (unless (or (timerp my-gptel-sessions--timer)
              (<= my-gptel-sessions-refresh-interval 0))
    (setq my-gptel-sessions--timer
          (run-with-timer my-gptel-sessions-refresh-interval
                          my-gptel-sessions-refresh-interval
                          #'my-gptel-sessions--refresh))))

(defun my-gptel-sessions ()
  "List all gptel sessions and the state of their requests."
  (interactive)
  (with-current-buffer (get-buffer-create my-gptel-sessions-buffer)
    (unless (derived-mode-p 'my-gptel-sessions-mode)
      (my-gptel-sessions-mode))
    (tabulated-list-print t)
    (pop-to-buffer (current-buffer))))
